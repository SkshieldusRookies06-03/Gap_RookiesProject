$ curl localhost:8080/actuator/health                               → 200 {"status":"UP"}
$ curl localhost:8080/actuator/metrics                              → 401 {"success":false,...,"statusCode":401,"code":"UNAUTHORIZED"}
$ curl localhost:8080/actuator/metrics -H "Authorization: Bearer $USER"  → 403 {"success":false,...,"statusCode":403,"code":"FORBIDDEN"}
$ curl localhost:8080/actuator/env                                  → 401 UNAUTHORIZED
